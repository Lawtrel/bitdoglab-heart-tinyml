cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico_w CACHE STRING "Pico W / RP2040")
if(NOT PICO_BOARD STREQUAL "pico_w" AND NOT PICO_BOARD STREQUAL "pico")
  message(FATAL_ERROR "Use RP2040: pico_w ou pico. Pico 2 não pertence a este experimento.")
endif()
set(PCG_APP demo CACHE STRING "Aplicação: demo ou benchmark")
set_property(CACHE PCG_APP PROPERTY STRINGS demo benchmark)
if(NOT PCG_APP MATCHES "^(demo|benchmark)$")
  message(FATAL_ERROR "PCG_APP deve ser demo ou benchmark")
endif()
if(NOT DEFINED PICO_SDK_PATH)
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
  message(FATAL_ERROR "Defina PICO_SDK_PATH para o Pico SDK 2.1.0")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(pcg_bitdoglab C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
set(PCG_GENERATED "${CMAKE_CURRENT_LIST_DIR}/generated" CACHE PATH "Modelo e coeficientes congelados")
set(PICO_TFLMICRO_PATH "$ENV{PICO_TFLMICRO_PATH}" CACHE PATH "Checkout pico-tflmicro")
if(NOT EXISTS "${PICO_TFLMICRO_PATH}/CMakeLists.txt")
  message(FATAL_ERROR "Defina PICO_TFLMICRO_PATH. Consulte docs/firmware.md")
endif()
foreach(name model_data.cpp model_data.h model_config.h frontend_config.h)
  if(NOT EXISTS "${PCG_GENERATED}/${name}")
    message(FATAL_ERROR "Modelo gerado ausente: ${name}")
  endif()
endforeach()
add_subdirectory("${PICO_TFLMICRO_PATH}" "${CMAKE_BINARY_DIR}/tflmicro" EXCLUDE_FROM_ALL)
target_compile_options(pico-tflmicro PRIVATE -Os -fno-rtti -fno-exceptions -fno-threadsafe-statics)
add_executable(pcg_bitdoglab "apps/${PCG_APP}/main.cpp" shared/frontend.cpp "${PCG_GENERATED}/model_data.cpp")
target_include_directories(pcg_bitdoglab PRIVATE shared "${PCG_GENERATED}")
target_link_libraries(pcg_bitdoglab PRIVATE pico-tflmicro pico_stdlib hardware_clocks)
if(PCG_APP STREQUAL "demo")
  target_link_libraries(pcg_bitdoglab PRIVATE hardware_i2c hardware_pwm hardware_adc hardware_dma)
endif()
target_compile_options(pcg_bitdoglab PRIVATE -O2 -fno-exceptions -fno-rtti -fstack-usage)
target_compile_definitions(pcg_bitdoglab PRIVATE PICO_STDIO_USB_DEFAULT_CRLF=0 PICO_STACK_SIZE=4096)
pico_enable_stdio_usb(pcg_bitdoglab 1)
pico_enable_stdio_uart(pcg_bitdoglab 0)
pico_add_extra_outputs(pcg_bitdoglab)
target_link_options(pcg_bitdoglab PRIVATE -Wl,--print-memory-usage)
